# Execute the QoE probe every 10 minutes, writing logs to the log directory
# and discarding cron stdout/stderr logs.
# Ensure INFLUXDB_TOKEN is configured securely via the local .env file generated by set_influx_token.py
*/10 * * * * /usr/bin/python3 /absolute/path/to/CX-RADAR/scripts/qoe_probe.py --config-path /absolute/path/to/CX-RADAR/config/probe-catalog.json >/dev/null 2>&1
